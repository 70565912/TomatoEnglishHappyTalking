// RETIRED REFERENCE ONLY: this tool targets the removed Web splitter.
import fs from 'node:fs';

import {
  splitOriginalSentencesV3,
  splitReadAloudV3Detailed,
} from '../src/readAloudSplitterV3';

const args = process.argv.slice(2);
const valueAfter = (name: string): string | undefined => {
  const index = args.indexOf(name);
  return index >= 0 ? args[index + 1] : undefined;
};

const filePath = valueAfter('--file');
const contains = valueAfter('--contains')?.toLowerCase();
if (!filePath) throw new Error('Usage: --file <english.txt> [--contains <text>]');

const source = fs.readFileSync(filePath, 'utf8');
for (const sentence of splitOriginalSentencesV3(source)) {
  if (contains && !sentence.toLowerCase().includes(contains)) continue;
  process.stdout.write(`${JSON.stringify({
    sentence,
    chunks: splitReadAloudV3Detailed(sentence),
  }, null, 2)}\n`);
}
