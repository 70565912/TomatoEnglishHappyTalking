export type SplitOptions = {

  targetMinWords?: number;
  targetMaxWords?: number;

type Token = {

  bare: string;
  start: number;
  end: number;
  tags: Set<string>;
};

type Boundary = {
  position: number;
  cost: number;
  quality: string;
  protected: boolean;
};

type PathState = {
  cost: number;
  previousPosition: number;
  previousLength: number;
};

export function splitReadAloudV2(text: string, options: SplitOptions = {}): string[] {
  const resolved = {
    maxWords: options.maxWords ?? HARD_MAX_WORDS,
    targetMinWords: options.targetMinWords ?? 10,
    targetMaxWords: options.targetMaxWords ?? 20,
    maxLineWidthPx: options.maxLineWidthPx ?? DEFAULT_MAX_LINE_WIDTH_PX,
    fontSizePx: options.fontSizePx ?? DEFAULT_FONT_SIZE_PX,
  };
  const output: string[] = [];
  for (const paragraph of normalizeArticleParagraphs(text)) {
    // Coarse detection is normalization only: it repairs glued OCR sentence
    // boundaries, then the entire paragraph goes through one global search.
    // Full stops remain high-quality candidate boundaries, never hard walls.
    const normalizedParagraph = coarseSentenceCandidates(paragraph).join(' ');
    output.push(...splitParagraphWithDp(normalizedParagraph, resolved));
  }
  return output.map((value) => value.replace(/\s+/gu, ' ').trim()).filter(Boolean);
}

export function normalizeForRoundTrip(text: string): string {
  return normalizeArticleParagraphs(text)
    .flatMap(coarseSentenceCandidates)
    .join(' ')
    .replace(/\s+/gu, ' ')
    .replace(/-\s+/gu, '-')
    .replace(/([—–])\s+/gu, '$1')
    .trim();
}

export function wordCount(text: string): number {
  return text.trim() ? text.trim().split(/\s+/u).length : 0;
}

export function isIncompleteShortChunk(text: string): boolean {
  const spokenUnits = text.trim().split(/\s+/u).reduce((count, token) => {
    const joinedUnits = token.split(/[-—–]/u).filter(Boolean).length;
    return count + Math.max(1, joinedUnits);
  }, 0);
  if (spokenUnits >= 6) return false;
  const trimmed = text.trim();
  if (FINAL_PUNCTUATION.test(trimmed) || STRONG_PUNCTUATION.test(trimmed)) return false;
  if (/[.!?…,]["”’]\s+(?:said|asked|replied|cried)\b[^.!?]*[,]$/iu.test(trimmed)) return false;
  if (/^(?:this\s+time,\s+)?at\s+last,$/iu.test(trimmed)) return false;
  const tokens = tagTokens(trimmed);
  return !(COMMA_PUNCTUATION.test(trimmed) && hasClause(tokens, 0, tokens.length));
}

/** Deterministic width from the bundled Nunito ExtraBold font advances. */
export function measureNunitoExtraBoldPx(text: string, fontSizePx = DEFAULT_FONT_SIZE_PX): number {
  const em = [...text].reduce(
    (total, character) => total + (NUNITO_EXTRA_BOLD_ADVANCE_EM[character] ?? NUNITO_EXTRA_BOLD_UNKNOWN_ADVANCE_EM),
    0,
  );
  return em * fontSizePx;
}

export function fitsEnglishSubtitleLine(text: string, options: SplitOptions = {}): boolean {
  return measureNunitoExtraBoldPx(text, options.fontSizePx ?? DEFAULT_FONT_SIZE_PX)
    <= (options.maxLineWidthPx ?? DEFAULT_MAX_LINE_WIDTH_PX);
}

function normalizeArticleParagraphs(text: string): string[] {
  const normalized = text
    .replace(/\r\n?/gu, '\n')
    .replace(/[‐‑‒]/gu, '-')
    .replace(/([A-Za-z])\s+-\s+(?=[A-Za-z])/gu, '$1-')
    .replace(/([—–])(?=[^\s"'”’])/gu, '$1 ')
    .replace(/[ \t]+/gu, ' ')
    .trim();
  if (!normalized) return [];

  const paragraphs: string[] = [];
  let current: string[] = [];
  const flush = () => {
    const value = current.join(' ').replace(/\s+/gu, ' ').trim();
    if (value && !isImportedHeadingLine(value)) paragraphs.push(value);
    current = [];
  };
  for (const rawLine of normalized.split('\n')) {
    const line = stripCjkPrefix(rawLine.trim());
    if (!line) {
      flush();
      continue;
    }
    if (isImportedHeadingLine(line)) {
      flush();
      continue;
    }
    current.push(line);
  }
  flush();
  return paragraphs;
}

function stripCjkPrefix(line: string): string {
  const firstLatin = line.search(/[A-Za-z"“'‘]/u);
  if (firstLatin > 0 && /[\p{Script=Han}]/u.test(line.slice(0, firstLatin))) {
    return line.slice(firstLatin).trim();
  }
  return line;
}

function isImportedHeadingLine(line: string): boolean {
  const value = line.trim();
  if (/^chapter\s+(?:[ivxlcdm]+|\d+)\s*[:：.-]?\s+.{1,90}$/iu.test(value)) return true;
  if (/^e\d{1,3}$/iu.test(value)) return true;
  if (/^[A-Z][A-Z '\-–—]{2,50}$/u.test(value)) return true;
  if (/^(?:chapter|book)\s+(?:[ivxlcdm]+|\d+)\s*[:：.-]?\s*$/iu.test(value)) return true;
  if (/^(?:chapter\s+)?\d+\s*[:：.-]\s*[^.!?]{1,90}$/iu.test(value)) return true;
  if (/^(?:the\s+wind\s+in\s+the\s+willows|alice'?s\s+adventures\s+in\s+wonderland)\s*[-–—:]\s*(?:e|episod(?:e)?)\s*\d+/iu.test(value)) return true;
  const words = value.split(/\s+/u);
  if (words.length <= 8 && !/[.!?]["”’)]*$/u.test(value)) {
    const titleWords = words.filter((word) => /^[A-Z][A-Za-z'’-]*$/u.test(word));
    if (titleWords.length >= Math.max(2, Math.ceil(words.length * 0.7))) return true;
  }
  return false;
}

function coarseSentenceCandidates(paragraph: string): string[] {
  const output: string[] = [];
  let start = 0;
  let quoteOpen = false;
  for (let index = 0; index < paragraph.length; index += 1) {
    const char = paragraph[index];
    if (char === '“') quoteOpen = true;
    else if (char === '”') quoteOpen = false;
    else if (char === '"') quoteOpen = !quoteOpen;
    if (!'.!?…'.includes(char)) continue;
    if (char === '.' && isDecimalPoint(paragraph, index)) continue;
    const current = paragraph.slice(start, index + 1);
    if (char === '.' && isProtectedPeriod(current)) continue;
    // Do not turn punctuation inside one still-open quotation into an
    // irreversible coarse boundary. DP can still use the punctuation later,
    // but it can also keep a multi-sentence utterance together.
    if (quoteOpen && !/^\s+["“'‘][A-Z]/u.test(paragraph.slice(index + 1))) continue;
    let end = index + 1;
    while (end < paragraph.length && /["'”’)}\]]/u.test(paragraph[end])) {
      const punctuation = paragraph[end];
      const nextAfterQuote = paragraph[end + 1] ?? '';
      // In glued `He left."She stayed`, the straight quote opens the next
      // sentence; it is not closing punctuation belonging to `left.`.
      if ((punctuation === '"' || punctuation === '“' || punctuation === '‘')
          && end === index + 1
          && /[A-Z]/u.test(nextAfterQuote)) break;
      end += 1;
    }
    let lookahead = end;
    while (lookahead < paragraph.length && /\s/u.test(paragraph[lookahead])) lookahead += 1;
    if (lookahead >= paragraph.length) {
      output.push(paragraph.slice(start, end).trim());
      start = lookahead;
      index = end - 1;
      continue;
    }
    const next = paragraph[lookahead];
    const glued = lookahead === end;
    const nextStartsSentence = /[A-Z0-9"“'‘(]/u.test(next);
    const attributionContinuation = /^[a-z]/u.test(next) && /["”’]$/u.test(paragraph.slice(start, end));
    if (nextStartsSentence && !attributionContinuation && (!glued || /["“'‘A-Z]/u.test(next))) {
      output.push(paragraph.slice(start, end).trim());
      start = lookahead;
      index = lookahead - 1;
    }
  }
  const tail = paragraph.slice(start).trim();
  if (tail) output.push(tail);
  return output;
}

function splitParagraphWithDp(paragraph: string, options: Required<SplitOptions>): string[] {
  const tokens = tagTokens(paragraph);
  if (tokens.length === 0) return [];
  if (tokens.length <= options.targetMaxWords) return [paragraph.trim()];
  const boundaries = buildBoundaries(paragraph, tokens);
  const byPosition = new Map(boundaries.map((boundary) => [boundary.position, boundary]));
  const states: Array<Map<number, PathState>> = Array.from({ length: tokens.length + 1 }, () => new Map());
  states[0].set(0, { cost: 0, previousPosition: -1, previousLength: -1 });

  for (let start = 0; start < tokens.length; start += 1) {
    if (states[start].size === 0) continue;
    for (let end = start + 1; end <= Math.min(tokens.length, start + options.maxWords); end += 1) {
      const text = paragraph.slice(tokens[start].start, tokens[end - 1].end).trim();
      const boundary = byPosition.get(end);
      if (!boundary) continue;
      const isFinal = end === tokens.length;
      const length = end - start;
      const completeClause = hasClause(tokens, start, end);
      const containsSemanticClauseBoundary = boundaries.some((candidate) =>
        candidate.position > start
        && candidate.position < end
        && candidate.quality === 'relative_clause');
      for (const [previousLength, state] of states[start]) {
        if (boundary.protected && !isFinal) continue;
        const splitCountCost = isFinal ? 0 : 60;
        const cost = state.cost
          + segmentCost(
            text,
            length,
            previousLength,
            isFinal,
            completeClause,
            containsSemanticClauseBoundary,
            options,
          )
          + splitCountCost
          + (isFinal ? 0 : boundary.cost);
        const existing = states[end].get(length);
        if (!existing || cost < existing.cost) {
          states[end].set(length, { cost, previousPosition: start, previousLength });
        }
      }
    }
  }

  if (states[tokens.length].size === 0) return emergencyWordSplit(paragraph, tokens, options);
  let finalLength = [...states[tokens.length]].sort((a, b) => a[1].cost - b[1].cost)[0][0];
  const spans: Array<[number, number]> = [];
  let position = tokens.length;
  while (position > 0) {
    const state = states[position].get(finalLength);
    if (!state) throw new Error('invalid DP backpointer');
    spans.push([state.previousPosition, position]);
    position = state.previousPosition;
    finalLength = state.previousLength;
  }
  spans.reverse();
  return spans.map(([start, end]) => paragraph.slice(tokens[start].start, tokens[end - 1].end).trim());
}

function segmentCost(
  text: string,
  length: number,
  previousLength: number,
  isFinal: boolean,
  completeClause: boolean,
  containsSemanticClauseBoundary: boolean,
  options: Required<SplitOptions>,
): number {
  const targetWords = (options.targetMinWords + options.targetMaxWords) / 2;
  // Segment length decides whether another chunk is worthwhile.  Boundary
  // punctuation is deliberately absent here: a full stop may be the best
  // place to split, but it must not create a reason to split by itself.
  let cost = (length - targetWords) ** 2;
  const endsCompleteBreath = FINAL_PUNCTUATION.test(text) || STRONG_PUNCTUATION.test(text);
  const firstWord = text.trim().split(/\s+/u)[0]
    ?.replace(/^["'“”‘’([{]+|[^A-Za-z'-]+$/gu, '')
    .toLowerCase();
  if (length < options.targetMinWords) {
    const shortfall = options.targetMinWords - length;
    cost += shortfall ** 2 * (endsCompleteBreath ? 8 : 20);
    if (!completeClause && !endsCompleteBreath) cost += shortfall * 30;
    if (firstWord && (COORDINATORS.has(firstWord) || PREPOSITIONS.has(firstWord))) {
      cost += shortfall * 40;
    }
  } else if (length > options.targetMaxWords) {
    const softOverflow = length - options.targetMaxWords;
    // One or two words above the 10-20 comfort range are cheaper than
    // manufacturing two weak breaths.  From 23 words onward the curve rises
    // sharply; 30 remains the absolute candidate limit.
    cost += softOverflow ** 2 * 2;
    if (length > options.targetMaxWords + 2) {
      cost += (length - options.targetMaxWords - 2) ** 2 * 15;
    }
    // A mildly overlength single clause may remain intact, but packing an
    // additional complete clause into it is a separate semantic reason to
    // split. This lets `...animals | whose residences...` win without making
    // punctuation itself an incentive to manufacture extra chunks.
    if (containsSemanticClauseBoundary) cost += 130;
  }
  if (firstWord
      && !completeClause
      && (COORDINATORS.has(firstWord) || PREPOSITIONS.has(firstWord))) {
    cost += 600;
  }
  if (length < 6 && !endsCompleteBreath) cost += (6 - length) ** 2 * 90;
  if (isFinal && length < 6 && !endsCompleteBreath) cost += 180;
  if (previousLength > 0) cost += Math.abs(length - previousLength);
  return cost;
}

function buildBoundaries(sentence: string, tokens: Token[]): Boundary[] {
  const boundaries: Boundary[] = [];
  for (let position = 1; position <= tokens.length; position += 1) {
    if (position === tokens.length) {
      boundaries.push({ position, cost: 0, quality: 'end', protected: false });
      continue;
    }
    const left = tokens[position - 1];
    const right = tokens[position];
    const protectedBoundary = isProtectedBoundary(tokens, position, sentence);
    const leftClause = hasClause(tokens, Math.max(0, position - 12), position);
    const rightClause = hasClause(tokens, position, Math.min(tokens.length, position + 12));
    // All boundary costs are non-negative.  They rank possible cut locations
    // after the segment score has established that an extra chunk is useful;
    // punctuation therefore cannot earn points merely by increasing chunks.
    let cost = 240;
    let quality = 'emergency';
    if (FINAL_PUNCTUATION.test(left.raw) && SPEECH_ATTRIBUTIONS.has(right.bare)) {
      cost = 5000;
      quality = 'speech_attribution';
    } else if (isDialogueBoundary(left, right, sentence)) {
      cost = 0;
      quality = 'dialogue';
    } else if (FINAL_PUNCTUATION.test(left.raw)) {
      cost = 0;
      quality = 'sentence';
    } else if (STRONG_PUNCTUATION.test(left.raw) && leftClause && rightClause) {
      cost = 15;
      quality = 'strong_clause';
    } else if (COMMA_PUNCTUATION.test(left.raw) && right.bare === 'only') {
      cost = 55;
      quality = 'comma_discourse';
    } else if (COMMA_PUNCTUATION.test(left.raw) && leftClause && rightClause) {
      cost = 35;
      quality = 'comma_clause';
    } else if (COMMA_PUNCTUATION.test(left.raw)
      && COORDINATORS.has(right.bare)
      && position + 1 < tokens.length
      && tokens[position + 1].tags.has('Gerund')) {
      cost = 45;
      quality = 'comma_participial';
    } else if (RELATIVE_SPLIT_STARTS.has(right.bare) && leftClause && rightClause) {
      cost = 40;
      quality = 'relative_clause';
    } else if (SUBORDINATORS.has(right.bare) && leftClause && rightClause) {
      cost = 55;
      quality = 'subordinate';
    } else if (COORDINATORS.has(right.bare)
      && leftClause
      && position + 1 < tokens.length
      && (DETERMINERS.has(tokens[position + 1].bare)
        || tokens[position + 1].tags.has('Noun')
        || tokens[position + 1].tags.has('ProperNoun'))) {
      cost = 60;
      quality = 'parallel_coord';
    } else if (COORDINATORS.has(right.bare) && leftClause && rightClause) {
      cost = 60;
      quality = 'coord_clause';
    } else if (COORDINATORS.has(right.bare)
      && leftClause
      && position + 1 < tokens.length
      && tokens[position + 1].tags.has('Verb')) {
      cost = 60;
      quality = 'parallel_predicate';
    } else if (COMMA_PUNCTUATION.test(left.raw)
      && (COORDINATORS.has(right.bare)
        || SUBORDINATORS.has(right.bare)
        || right.tags.has('Gerund')
        || right.tags.has('PresentTense'))) {
      cost = 70;
      quality = 'comma_continuation';
    } else if (COMMA_PUNCTUATION.test(left.raw)) {
      cost = 80;
      quality = 'comma';
    } else if (STRONG_PUNCTUATION.test(left.raw)) {
      cost = 20;
      quality = 'strong';
    }
    boundaries.push({ position, cost, quality, protected: protectedBoundary });
  }
  return boundaries;
}

function isProtectedBoundary(tokens: Token[], position: number, sentence: string): boolean {
  const left = tokens[position - 1];
  const right = tokens[position];
  const lt = left.tags;
  const rt = right.tags;
  if (!left || !right) return false;
  if ((DETERMINERS.has(left.bare) || lt.has('Determiner') || lt.has('Possessive'))
      && /^["“'‘]/u.test(right.raw)) {
    return true;
  }
  if (COMMA_PUNCTUATION.test(left.raw)
      && SPEECH_ATTRIBUTIONS.has(left.bare)
      && !/^["“'‘]/u.test(right.raw)) {
    return true;
  }
  if (FINAL_PUNCTUATION.test(left.raw) && SPEECH_ATTRIBUTIONS.has(right.bare)) {
    return true;
  }
  if (FINAL_PUNCTUATION.test(left.raw) && /^[a-z]/u.test(right.raw)) {
    return true;
  }
  if (COMMA_PUNCTUATION.test(left.raw)
      && /["”’]$/u.test(left.raw)
      && SUBJECT_PRONOUNS.has(right.bare)
      && position + 1 < tokens.length
      && SPEECH_ATTRIBUTIONS.has(tokens[position + 1].bare)) {
    return true;
  }
  // A relative/subordinate clause starts a new constituent even when the
  // shallow tagger has extended the preceding NP/PP across its pronoun.
  // Example: `owned by animals | whose residences ...`.
  if (RELATIVE_SPLIT_STARTS.has(right.bare)
      && !PREPOSITIONS.has(left.bare)
      && !COORDINATORS.has(left.bare)
      && hasClause(tokens, Math.max(0, position - 12), position)) {
    return false;
  }
  const leftClause = hasClause(tokens, Math.max(0, position - 12), position);
  const rightStartsParticiple = rt.has('Gerund')
    || rt.has('PastTense')
    || /(?:ing|ed)$/u.test(right.bare);
  if (COMMA_PUNCTUATION.test(left.raw)
      && leftClause
      && rightStartsParticiple) {
    return true;
  }
  // Visible pause punctuation outranks local POS adjacency.  For example,
  // `the Mole, waving ...` must remain a legal breath boundary even though
  // compromise tags Mole as a noun and waving as a verb.
  if (/[,;:!?—–]["'”’)}\]]*$/u.test(left.raw)) return false;
  // Never strand a function head at the end of a subtitle.  The more precise
  // span checks below protect longer constituents, while these adjacency
  // guards cover tagging misses such as `at | once`, `in a | hesitating` and
  // list continuations ending in `and | pots`.
  if (DETERMINERS.has(left.bare)) return true;
  if (/-$/u.test(left.raw) || /^-/u.test(right.raw)) return true;
  if (PREPOSITIONS.has(left.bare)) return true;
  if (COORDINATORS.has(left.bare)) return true;
  if (AUXILIARIES.has(left.bare)) return true;
  if (SPEECH_ATTRIBUTIONS.has(left.bare)) return true;
  if (left.tags.has('Verb') && (right.bare === 'as' || right.bare === 'that')) return true;
  if (left.tags.has('Verb')
      && right.bare === 'and'
      && position + 1 < tokens.length
      && tokens[position + 1].tags.has('Verb')) {
    return true;
  }
  if (right.bare === 'or'
      && (lt.has('Noun') || lt.has('ProperNoun'))
      && position + 1 < tokens.length
      && (DETERMINERS.has(tokens[position + 1].bare) || tokens[position + 1].tags.has('Determiner'))) {
    return true;
  }
  if (left.tags.has('Verb') && (PREPOSITIONS.has(right.bare) || right.tags.has('Preposition'))) return true;
  if (left.tags.has('Adjective') && (PREPOSITIONS.has(right.bare) || right.tags.has('Preposition'))) return true;
  if (left.tags.has('Verb') && PHRASAL_PARTICLES.has(right.bare)) return true;
  if (entityTag(lt) && sameEntityKind(lt, rt)) return true;
  if ((DETERMINERS.has(left.bare) || lt.has('Determiner')) && (rt.has('Adjective') || rt.has('Noun') || rt.has('ProperNoun') || right.bare === 'only')) return true;
  if ((lt.has('Adjective') || left.bare === 'only') && (rt.has('Noun') || rt.has('ProperNoun'))) return true;
  if ((lt.has('Noun') || lt.has('ProperNoun')) && (rt.has('Noun') || rt.has('ProperNoun'))) return true;
  if ((AUXILIARIES.has(left.bare) || lt.has('Auxiliary') || lt.has('Modal')) && (rt.has('Adverb') || rt.has('Verb'))) return true;
  if (lt.has('Adverb') && rt.has('Verb') && position >= 2 && (tokens[position - 2].tags.has('Auxiliary') || AUXILIARIES.has(tokens[position - 2].bare))) return true;
  if ((PREPOSITIONS.has(left.bare) || lt.has('Preposition')) && (DETERMINERS.has(right.bare) || rt.has('Pronoun') || rt.has('Noun') || rt.has('ProperNoun'))) return true;
  if ((SUBJECT_PRONOUNS.has(left.bare) || lt.has('ProperNoun') || lt.has('Noun')) && (rt.has('Verb') || rt.has('Modal') || rt.has('Auxiliary'))) return true;
  if (left.bare === 'to' && (rt.has('Verb') || right.bare === 'be')) return true;
  if (COORDINATORS.has(left.bare) && (PREPOSITIONS.has(right.bare) || SUBORDINATORS.has(right.bare))) return true;
  if (insideDeterminerNounPhrase(tokens, position)) return true;
  if (insidePrepositionalObject(tokens, position)) return true;
  return false;
}

function insideDeterminerNounPhrase(tokens: Token[], position: number): boolean {
  for (let start = Math.max(0, position - 5); start < position; start += 1) {
    const first = tokens[start];
    if (!(DETERMINERS.has(first.bare) || first.tags.has('Determiner') || first.tags.has('Possessive'))) continue;
    const span = tokens.slice(start + 1, Math.min(tokens.length, position + 1));
    if (span.length === 0 || span.length > 5) continue;
    if (span.some((token) => /[.!?,;:—–]/u.test(token.raw))) continue;
    if (span.every((token) =>
      token.tags.has('Adjective')
      || token.tags.has('Adverb')
      || token.tags.has('Noun')
      || token.tags.has('ProperNoun')
      || (token.tags.size === 0 && /^[A-Za-z'’-]+$/u.test(token.raw))
      || token.bare === 'only')) return true;
  }
  return false;
}

function insidePrepositionalObject(tokens: Token[], position: number): boolean {
  for (let start = Math.max(0, position - 6); start < position; start += 1) {
    const first = tokens[start];
    if (!(PREPOSITIONS.has(first.bare) || first.tags.has('Preposition'))) continue;
    const span = tokens.slice(start + 1, Math.min(tokens.length, position + 1));
    if (span.length === 0 || span.length > 6) continue;
    if (span.some((token) => /[.!?,;:—–]/u.test(token.raw))) continue;
    if (span.every((token) =>
      DETERMINERS.has(token.bare)
      || token.tags.has('Determiner')
      || token.tags.has('Possessive')
      || token.tags.has('Adjective')
      || token.tags.has('Adverb')
      || token.tags.has('Noun')
      || token.tags.has('ProperNoun')
      || token.tags.has('Pronoun'))) return true;
  }
  return false;
}

function entityTag(tags: Set<string>): string | null {
  for (const tag of ['Person', 'Place', 'Organization', 'ProperNoun']) if (tags.has(tag)) return tag;
  return null;
}

function sameEntityKind(left: Set<string>, right: Set<string>): boolean {
  return ['Person', 'Place', 'Organization', 'ProperNoun'].some((tag) => left.has(tag) && right.has(tag));
}

function hasClause(tokens: Token[], start: number, end: number): boolean {
  let subjectSeen = false;
  for (let index = start; index < end; index += 1) {
    const token = tokens[index];
    if (token.tags.has('Noun') || token.tags.has('ProperNoun') || token.tags.has('Pronoun') || SUBJECT_PRONOUNS.has(token.bare)) {
      subjectSeen = true;
      continue;
    }
    if (subjectSeen && (token.tags.has('Verb') || token.tags.has('Modal') || token.tags.has('Auxiliary'))) return true;
  }
  return false;
}

function isDialogueBoundary(left: Token, right: Token, sentence: string): boolean {
  const before = sentence.slice(0, left.end);
  return /[.!?]["”’]$/u.test(left.raw)
    || (FINAL_PUNCTUATION.test(left.raw) && /^["“'‘]/u.test(right.raw))
    || (/["”’]$/u.test(left.raw) && /^["“'‘]/u.test(right.raw))
    || (/[,:—–]$/u.test(left.raw) && /^["“'‘]/u.test(right.raw) && !insideUnclosedQuote(before, before.length));
}

function insideUnclosedQuote(text: string, offset: number): boolean {
  const prefix = text.slice(0, offset);
  let open = false;
  for (const char of prefix) {
    if (char === '“') open = true;
    else if (char === '”') open = false;
    else if (char === '"') open = !open;
  }
  return open;
}

function tagTokens(text: string): Token[] {
  const rawTokens = [...text.matchAll(/\S+/gu)].flatMap((match) => {
    const raw = match[0];
    const start = match.index ?? 0;
    if ((raw.match(/-/gu) ?? []).length < 6) {
      return [{ raw, bare: normalizeBare(raw), start, end: start + raw.length, tags: new Set<string>() }];
    }
    // A few Victorian/OCR compounds contain a whole clause joined by hyphens.
    // Hyphens become legal virtual boundaries so the English line remains
    // representable; round-trip normalization ignores whitespace after `-`.
    const pieces: Token[] = [];
    for (const piece of raw.matchAll(/[^-]+-?|-/gu)) {
      const value = piece[0];
      const pieceStart = start + (piece.index ?? 0);
      if (value === '-') continue;
      pieces.push({ raw: value, bare: normalizeBare(value), start: pieceStart, end: pieceStart + value.length, tags: new Set<string>() });
    }
    return pieces;
  });
  const terms = (nlp(text).terms().json() as Array<{ terms?: Array<{ normal?: string; tags?: string[] }> }> )
    .flatMap((entry) => entry.terms ?? []);
  let termIndex = 0;
  for (const token of rawTokens) {
    while (termIndex < terms.length) {
      const term = terms[termIndex];
      termIndex += 1;
      const normal = normalizeBare(term.normal ?? '');
      if (!normal) continue;
      if (token.bare === normal || token.bare.includes(normal) || normal.includes(token.bare)) {
        token.tags = new Set(term.tags ?? []);
        break;
      }
    }
  }
  return rawTokens;
}

function normalizeBare(value: string): string {
  return value.toLowerCase().replace(TRAILING_QUOTES, '').replace(/^["'“‘([{]+/u, '').replace(/[^a-z0-9’-]+$/u, '');
}

function emergencyWordSplit(sentence: string, tokens: Token[], options: Required<SplitOptions>): string[] {
  const chunks: string[] = [];
  let start = 0;
  while (start < tokens.length) {
    const chosen = Math.min(tokens.length, start + options.maxWords);
    chunks.push(sentence.slice(tokens[start].start, tokens[chosen - 1].end).trim());
    start = chosen;
  }
  return chunks;
}

function isDecimalPoint(text: string, index: number): boolean {
  return index > 0 && index + 1 < text.length && /\d/u.test(text[index - 1]) && /\d/u.test(text[index + 1]);
}

function isProtectedPeriod(current: string): boolean {
  const lower = current.toLowerCase();
  return [...ABBREVIATIONS].some((abbr) => lower.endsWith(abbr)) || /(?:\b[A-Z]\.){1,4}$/u.test(current);
}
