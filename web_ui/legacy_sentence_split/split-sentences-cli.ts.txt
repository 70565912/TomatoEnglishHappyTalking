// RETIRED REFERENCE ONLY: this tool targets the removed Web splitter.
import { readFile, writeFile } from 'node:fs/promises';
import path from 'node:path';
import process from 'node:process';
import {
  READ_ALOUD_SPLIT_VERSION,
  fitsEnglishSubtitleLine,
  maxUnpunctuatedWordCount,
  splitReadAloudV3,
  wordCount,
} from '../src/readAloudSplitterV3';

function requiredArgument(name: string): string {
  const index = process.argv.indexOf(name);
  const value = index >= 0 ? process.argv[index + 1] : undefined;
  if (!value) throw new Error(`missing ${name}`);
  return path.resolve(value);
}

const input = requiredArgument('--input');
const output = requiredArgument('--output');
const text = await readFile(input, 'utf8');
const sentences = splitReadAloudV3(text);
await writeFile(output, `${sentences.join('\n\n')}\n`, 'utf8');
const jsonIndex = process.argv.indexOf('--json');
if (jsonIndex >= 0 && process.argv[jsonIndex + 1]) {
  await writeFile(
    path.resolve(process.argv[jsonIndex + 1]),
    `${JSON.stringify({
      splitter: READ_ALOUD_SPLIT_VERSION,
      sentences: sentences.map((sentence, index) => ({
        index,
        text: sentence,
        wordCount: wordCount(sentence),
        maxUnpunctuatedWords: maxUnpunctuatedWordCount(sentence),
        widthFits: fitsEnglishSubtitleLine(sentence),
      })),
    }, null, 2)}\n`,
    'utf8',
  );
}
